import requests
from typing import List, Dict

class ProxyManager:
    def __init__(self):
        self.proxies = self.fetch_proxies()

    async def get_proxy(self) -> str:
        return self.proxies.pop(0)

    async def fetch_proxies(self) -> List[str]:
        response = requests.get('http://proxy.com/api/proxy-list')
        proxies_list = response.json()
        return [proxy['ip:port'] for proxy in proxies_list]
