**Pregunta 2: Código para scrapers/facebook_mcp.py**

Aquí tienes un ejemplo básico de cómo podrías configurar un scraper para Fa[2D[K
Facebook usando Kafka como MCP:

```python
import json
from kafka import KafkaProducer
from bs4 import BeautifulSoup
import requests

# Configuración Kafka
KAFKA_BROKER = 'localhost:9092'
KAFKA_TOPIC = 'facebook_updates'

def scrape_facebook(url):
    response = requests.get(url)
    if response.status_code == 200:
        soup = BeautifulSoup(response.text, 'html.parser')
        # Aquí puedes extraer la información que necesites
        posts = []
        for post in soup.find_all('div', class_='user-post'):
            title = post.find('h3').text
            content = post.find('p').text
            posts.append({'title': title, 'content': content})
        return json.dumps(posts)
    else:
        return None

def send_to_mcp(data):
    producer = KafkaProducer(bootstrap_servers=KAFKA_BROKER,
                             value_serializer=lambda v: v.encode('utf-8'))
    producer.send(KAFKA_TOPIC, value=data)
    producer.flush()

if __name__ == '__main__':
    url = 'https://www.facebook.com/example_page'
    data = scrape_facebook(url)
    if data:
        send_to_mcp(data)
```
