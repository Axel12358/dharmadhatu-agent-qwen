import asyncio

def generar_queries_facebook(config):
    """Genera combinaciones de países, subgéneros, tipos y ciudades."""
    queries = []
    paises = config.get("priorizar_paises", [])
    subgeneros = config.get("subgeneros", ["psytrance"])
    tipos = config.get("tipos_evento", ["festival"])
    ciudades = config.get("ciudades", [])

    if paises:
        for pais in paises:
            for sub in subgeneros:
                for tipo in tipos:
                    queries.append(f"{sub} {tipo} {pais}")
            if ciudades:
                for ciudad in ciudades:
                    queries.append(f"psytrance festival {ciudad} {pais}")
    else:
        for sub in subgeneros:
            for tipo in tipos:
                queries.append(f"{sub} {tipo}")
        if ciudades:
            for ciudad in ciudades:
                queries.append(f"psytrance festival {ciudad}")

    if not queries:
        queries.append("psytrance festival")

    max_queries = config.get("busquedas_facebook", 150)
    random.shuffle(queries)
    return queries[:max_queries]

async def scrape_una_query(query, page, timeout):
    """Scrapea una query de Facebook y devuelve eventos."""
    eventos = []
    try:
        url = f"https://www.facebook.com/events/search/?q={query.replace(' ', '+')}"
        await page.goto(url, timeout=timeout * 1000, wait_until='networkidle')
        await asyncio.sleep(random.uniform(2, 4))

        # Scroll
        for _ in range(3):
            await page.evaluate('window.scrollBy(0, window.innerHeight)')
            await asyncio.sleep(random.uniform(1, 2))

        selectores = [
            'div[role="article"]',
            'div[data-testid="event-card"]',
            'div[class*="event"]'
        ]

        elementos = []
        for selector in selectores:
            try:
                elementos = await page.query_selector_all(selector)
                if elementos:
                    break
            except:
                continue

        if not elementos:
            return eventos

        for elem in elementos:
            try:
                # Nombre
                nombre = "Sin nombre"
                for n_sel in ['span[dir="auto"]', 'span[class*="title"]', 'h2']:
                    nombre_elem = await elem.query_selector(n_sel)
                    if nombre_elem:
                        nombre = await nombre_elem.inner_text()
                        if nombre and nombre.strip() and nombre != "Sin nombre":
                            break

                # Fecha
                fecha = "Fecha no disponible"
                for f_sel in ['span[data-testid="event-card-date"]', 'span[class*="date"]']:
                    fecha_elem = await elem.query_selector(f_sel)
                    if fecha_elem:
                        fecha = await fecha_elem.inner_text()
                        if fecha and fecha.strip():
                            break

                # Lugar
                lugar = "Lugar no disponible"
                for l_sel in ['span[data-testid="event-card-location"]', 'span[class*="location"]']:
                    lugar_elem = await elem.query_selector(l_sel)
                    if lugar_elem:
                        lugar = await lugar_elem.inner_text()
                        if lugar and lugar.strip():
                            break
